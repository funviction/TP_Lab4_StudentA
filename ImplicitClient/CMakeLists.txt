cmake_minimum_required(VERSION 3.16)
project(ImplicitClient CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ImplicitApp main.cpp)
target_include_directories(ImplicitApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../3rdParty/PartnerSDK/include)
set(PARTNER_LIB_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../3rdParty/PartnerSDK/lib/NumericalDiff.lib")
target_link_libraries(ImplicitApp PRIVATE ${PARTNER_LIB_PATH})
add_custom_command(TARGET ImplicitApp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../3rdParty/PartnerSDK/bin/NumericalDiff.dll" $<TARGET_FILE_DIR:ImplicitApp> COMMENT "Copying DLL")
